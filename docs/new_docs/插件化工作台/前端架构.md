# 插件化工作台 - 前端架构

## 技术方案：Host SDK

宿主暴露 Vue 运行时，插件通过全局变量获取，不需要第三方依赖。

---

## 宿主端实现

### Host SDK 定义

```typescript
// employee-web/src/core/host-sdk.ts
import * as Vue from 'vue';
import * as VueRouter from 'vue-router';
import * as Pinia from 'pinia';

export interface HostSdk {
  Vue: typeof Vue;
  VueRouter: typeof VueRouter;
  Pinia: typeof Pinia;
  plugins: Map<string, PluginRegistration>;
  register(id: string, plugin: PluginRegistration): void;
  getPlugin(id: string): PluginRegistration | undefined;
}

export interface PluginRegistration {
  id: string;
  name: string;
  component: () => Promise<any>;
}

export function initHostSdk(): void {
  (window as any).__WORKFORCE_HOST__ = {
    Vue,
    VueRouter,
    Pinia,
    plugins: new Map(),
    register(id: string, plugin: PluginRegistration) {
      this.plugins.set(id, plugin);
    },
    getPlugin(id: string) {
      return this.plugins.get(id);
    },
  };
}
```

### 应用入口

```typescript
// employee-web/src/main.ts
import { initHostSdk } from './core/host-sdk';
initHostSdk();
// ...原有启动逻辑
```

### 插件加载器

```typescript
// employee-web/src/core/plugin-loader.ts
import { employeeApi } from '@/api';

const isDev = import.meta.env.DEV;

export async function loadPlugin(pluginId: string, integrity?: string): Promise<void> {
  const url = isDev
    ? `/plugins/${pluginId}.js`  // 开发模式：通过代理访问插件dev server
    : `/api/plugin-assets/${pluginId}/index.js`;  // 生产模式：后端提供

  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = url;

    // SRI校验（生产模式）
    if (!isDev && integrity) {
      script.integrity = integrity;
      script.crossOrigin = 'anonymous';
    }

    script.onload = resolve;
    script.onerror = reject;
    document.head.appendChild(script);
  });
}

export async function loadAllPlugins(): Promise<void> {
  // 调用后端API获取插件清单
  const response = await employeeApi.get('/api/employee/plugins/mine');

  // 处理 R<T> 响应格式
  if (response.code !== 0) {
    console.error('加载插件清单失败:', response.message);
    return;
  }

  const plugins = response.data;

  for (const plugin of plugins) {
    try {
      await loadPlugin(plugin.id, plugin.integrity);
    } catch (error) {
      console.error(`加载插件 ${plugin.id} 失败:`, error);
    }
  }
}
```

## 插件端实现

### 插件入口

```typescript
// plugins/leave/src/index.ts
const { ref, computed } = (window as any).__WORKFORCE_HOST__.Vue;

// 导入组件（静态导入，不使用动态import）
import LeaveView from './LeaveView.vue';

// 注册插件
const host = (window as any).__WORKFORCE_HOST__;
if (host) {
  host.register('leave', {
    id: 'leave',
    name: '假期管理',
    // 懒加载：返回Promise，但不是动态import()
    component: () => Promise.resolve(LeaveView),
  });
}
```

### 构建配置

```typescript
// plugins/leave/vite.config.ts
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';

export default defineConfig({
  plugins: [vue()],
  build: {
    lib: {
      entry: './src/index.ts',
      formats: ['iife'],
      name: 'PluginLeave',
      fileName: 'index',
    },
    rollupOptions: {
      // 外部化宿主提供的依赖
      external: [],
      output: {
        // 不使用globals映射，因为IIFE格式下Vue通过window.__WORKFORCE_HOST__.Vue访问
        // 所有依赖都从宿主SDK获取
        globals: {},
        // 确保IIFE正确导出
        exports: 'named',
      },
    },
    // 确保所有代码打包到一个文件（不拆分chunk）
    cssCodeSplit: false,
    sourcemap: false,
  },
});
```

**重要**：插件代码中不要直接 import vue，而是从 `window.__WORKFORCE_HOST__.Vue` 获取：

```typescript
// plugins/leave/src/LeaveView.vue
<script setup lang="ts">
// 不要这样：
// import { ref } from 'vue'

// 要这样：
const { ref, computed } = (window as any).__WORKFORCE_HOST__.Vue;

const count = ref(0);
</script>
```

## 调试方案

### 启动顺序

```bash
# 终端1：启动宿主
cd employee-web
npm run dev  # → http://localhost:5174

# 终端2：启动插件（开发模式输出到指定目录）
```

## 环境差异

| 环境 | 插件来源                     | 热更新 | SRI校验 |
|------|------------------------------|--------|---------|
| 开发 | /plugins/xxx.js (代理)       | 支持   | 无      |
| 生产 | /api/plugin-assets/xxx.js    | 不支持 | 有      |

---

## 注意事项

1. **不要使用动态import()**：IIFE格式不支持代码分包
2. **不要直接import vue**：从window.__WORKFORCE_HOST__.Vue获取
3. **插件代码必须打包成单个文件**：不能有外部依赖
4. **CSS需要命名空间**：避免样式冲突
5. **事件监听需要清理**：插件停用时移除所有监听器
